        for (encrypted in listOf(false, true)) {
            val archive = fixture(version = 1, code = 84, encrypted = encrypted, scope = null)
            val entries = wireEntries(archive, encrypted)
            val manifest = officialWireJson.parseToJsonElement(entries.getValue("manifest.json")).jsonObject
            val preferences = officialWireJson.parseToJsonElement(entries.getValue("preferences.json")).jsonObject
            assertEquals(setOf("formatVersion", "appVersionCode", "appVersionName", "createdAt", "sourceFilesRoot", "counts", "fileCount", "totalBytes", "preferencesSha256"), manifest.keys)
            assertEquals(setOf("novelAiToken", "fishAudioKey", "communitySession", "brushSize", "brushType"), preferences.keys)
            officialWireJson.decodeFromString<Official145Manifest>(manifest.toString())
            officialWireJson.decodeFromString<Official145Preferences>(preferences.toString())
        }
    }
    private fun wireEntries(archive: ByteArray, encrypted: Boolean): Map<String, String> =
        ZipInputStream(Official147FixtureCodec.decrypting(archive.inputStream(), if (encrypted) password else null)).use { zip ->
            buildMap { while (true) { val entry = zip.nextEntry ?: break; put(entry.name, zip.readBytes().toString(Charsets.UTF_8)); zip.closeEntry() } }
        }
    private val inspector = DesktopBackupReadOnlyInspector()
    private val password get() = "synthetic-password".toCharArray()
    private val emptyScope = InspectionScope(emptySet(), 0, 0, 0, 0, 0, false, false, true, false)
    private fun bytes(value: String) = value.toByteArray()
    private fun fixture(version: Int = 2, code: Int = 86, encrypted: Boolean = false,
        files: Map<String, ByteArray> = emptyMap(), scope: InspectionScope? = emptyScope,
        prefs: InspectionPreferences = InspectionPreferences(), manifestChange: (JsonObject) -> JsonObject = { it },
        wrongHash: Boolean = false, entryChange: (String) -> String = { it }): ByteArray {
        val out = ByteArrayOutputStream()
        ZipOutputStream(Official147FixtureCodec.encrypting(out, if (encrypted) password else null, version)).use { zip ->
            fun entry(name: String, data: ByteArray) { zip.putNextEntry(ZipEntry(entryChange(name))); zip.write(data); zip.closeEntry() }
            files.forEach { (name, data) -> entry("files/$name", data) }
            val preferences = bytes(backupJson.encodeToString(prefs))
            entry("preferences.json", preferences)
            entry("inventory.jsonl", bytes(files.map { (name, data) -> backupJson.encodeToString(AppBackupFileRecord(name, data.size.toLong(),
                if (wrongHash) "0".repeat(64) else DesktopBackupPolicy.digest(data))) }.joinToString("\n")))
            val manifest = InspectionManifest(version, code, "synthetic-official-format", 1, "synthetic-source", files.keys
                .filter { it.startsWith("entities/") }.groupingBy(DesktopBackupPolicy::entityType).eachCount(), files.size,
                files.values.sumOf { it.size.toLong() }, DesktopBackupPolicy.digest(preferences), scope)
            entry("manifest.json", bytes(manifestChange(backupJson.encodeToJsonElement(manifest).jsonObject).toString()))
        }
        return out.toByteArray()
    }
    private suspend fun <T> isolated(data: ByteArray, block: suspend (Path, Path) -> T): T {
        val root = Files.createTempDirectory("s0-synthetic-")
        try {
            val input = root.resolve("source.cbbackup"); Files.write(input, data)
            val live = Files.createDirectories(root.resolve("live")); Files.writeString(live.resolve("journal"), "synthetic-existing-evidence")
            val before = Files.walk(root).use { paths -> paths.filter { Files.isRegularFile(it) }.toList().associate { it.toString() to DesktopBackupPolicy.digest(Files.readAllBytes(it)) } }
            try { return block(input, root) } finally {
                val after = Files.walk(root).use { paths -> paths.filter { Files.isRegularFile(it) }.toList().associate { it.toString() to DesktopBackupPolicy.digest(Files.readAllBytes(it)) } }
                assertEquals(before, after)
            }
        } finally { root.toFile().deleteRecursively() }
    }
    @Test fun `official v1 and v2 plaintext and encrypted formats authenticate without restore admission`() = runBlocking<Unit> {
        for (version in 1..2) for (encrypted in listOf(false, true)) for (code in if (version == 1) listOf(84, 85, 86) else listOf(85, 86)) {
            val data = fixture(version, code, encrypted, scope = if (version == 1) null else emptyScope)
            isolated(data) { input, _ ->
                val result = inspector.inspect(input, if (encrypted) password else null)
                assertEquals(version, result.envelopeVersion); assertEquals(code, result.appVersionCode)
                assertTrue(result.formatReadable); assertFalse(result.restoreSupported); assertFalse(result.fullPortabilityAccepted)
                assertEquals(DesktopBackupPolicy.digest(data), result.sourceSha256)
                if (version == 2) assertEquals(DesktopBackupReadOnlyInspector.RestoreAssessment.UNSUPPORTED, result.restoreAssessment)
                println("fixture v$version code$code encrypted=$encrypted sha256=${result.sourceSha256} length=${data.size}")
            }
        }
    }

    @Test fun `exact duplicate zip names and CRC tampering fail closed`() = runBlocking<Unit> {
        val original = fixture(files = mapOf("images/a" to byteArrayOf(1), "images/b" to byteArrayOf(2)))
        val duplicate = original.copyOf()
        val needle = bytes("files/images/b")
        for (i in 0..duplicate.size - needle.size) if (needle.indices.all { duplicate[i + it] == needle[it] }) duplicate[i + needle.lastIndex] = 'a'.code.toByte()
        isolated(duplicate) { input, _ -> assertFailsWith<DesktopBackupReadOnlyInspector.InspectionFailure> { inspector.inspect(input, null) } }
        val corrupt = original.copyOf()
        // Corrupt the central-directory CRC while leaving its local record untouched.
        val central = (0..corrupt.size - 4).first { corrupt[it] == 0x50.toByte() && corrupt[it + 1] == 0x4b.toByte() && corrupt[it + 2] == 1.toByte() && corrupt[it + 3] == 2.toByte() }
        corrupt[central + 16] = (corrupt[central + 16].toInt() xor 1).toByte()
        isolated(corrupt) { input, _ -> assertFailsWith<DesktopBackupReadOnlyInspector.InspectionFailure> { inspector.inspect(input, null) } }
    }

    @Test fun `hard child JVM exit after authentication leaves source and synthetic root unchanged`() = runBlocking<Unit> {
        isolated(fixture(encrypted = true)) { input, _ ->
            val types = listOf(S0ReadOnlyCrashChild::class.java, DesktopBackupReadOnlyInspector::class.java,
                com.example.chatbar.data.local.entity.AppSettings::class.java, Unit::class.java,
                kotlinx.coroutines.Job::class.java, Json::class.java, kotlinx.serialization.KSerializer::class.java,
                com.sun.jna.Native::class.java, com.sun.jna.platform.win32.Kernel32::class.java,
                com.google.crypto.tink.KeysetHandle::class.java, com.google.crypto.tink.proto.Keyset::class.java,
                com.google.crypto.tink.proto.Keyset::class.java.superclass)
            val cp = types.map { Path.of(it.protectionDomain.codeSource.location.toURI()).toString() }.distinct().joinToString(File.pathSeparator)
            val java = Path.of(System.getProperty("java.home"), "bin", if (System.getProperty("os.name").startsWith("Windows")) "java.exe" else "java")
            val process = ProcessBuilder(java.toString(), "-Xmx128m", "-cp", cp, S0ReadOnlyCrashChild::class.java.name, input.toString())
                .redirectErrorStream(true).start()
            assertTrue(process.waitFor(60, TimeUnit.SECONDS))
            val output = process.inputStream.bufferedReader().readText()
            assertEquals(73, process.exitValue(), output)
            assertTrue(inspector.inspect(input, password).formatReadable)
            println("S0 synthetic child exit=${process.exitValue()} alive=${process.isAlive}")
        }
    }
    @Test fun `oversized source is refused before payload allocation`() = runBlocking<Unit> {
        val root = Files.createTempDirectory("s0-budget-")
        try {
            val input = root.resolve("synthetic-large.cbbackup")
            RandomAccessFile(input.toFile(), "rw").use { it.setLength(DesktopBackupReadOnlyInspector.MAX_COMPRESSED.toLong() + 1) }
            val error = assertFailsWith<DesktopBackupReadOnlyInspector.InspectionFailure> { inspector.inspect(input, null) }
            assertEquals("INPUT", error.stage)
            assertEquals(DesktopBackupReadOnlyInspector.MAX_COMPRESSED.toLong() + 1, Files.size(input))
            assertEquals(1, Files.list(root).use { it.count() }.toInt())
        } finally { root.toFile().deleteRecursively() }
    }
    @Test fun `cancellation has no work directories or retained files`() = runBlocking<Unit> {
        isolated(fixture(encrypted = true)) { input, _ ->
            val job = Job()
            var authenticated = false
            val cancellable = DesktopBackupReadOnlyInspector { authenticated = true; job.cancel() }
            assertFailsWith<CancellationException> { withContext(job) { cancellable.inspect(input, password) } }
            assertTrue(authenticated); assertTrue(job.isCancelled)
            // A second read proves the source handle was released after cancellation.
            assertTrue(inspector.inspect(input, password).formatReadable)
        }
    }
}

    }
